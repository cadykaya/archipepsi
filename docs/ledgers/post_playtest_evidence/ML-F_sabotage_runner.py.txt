"""ML-F1/F2 sabotages: break one rule, confirm a named failure, restore.

Same protocol as the D-9, H-STATUS-2 and D-10 runners: each row edits one
file in the wt-life work tree, runs `make godot-enemy-footing`, records
every FAIL line, and restores the file byte for byte (sha256 checked). A
row is CAUGHT only when the check it names is among the failures. Run
only when no other Godot suite is running.
"""
from __future__ import annotations

import hashlib
import os
import subprocess
import sys
from pathlib import Path

TREE = Path(os.environ.get("MLF_TREE",
                           Path(__file__).resolve().parent / "wt-life"))
ZC = "godot/scripts/gameplay/zone_controller.gd"
EF = "godot/scripts/enemies/enemy_footing.gd"
EN = "godot/scripts/enemies/enemy.gd"
TARGET = "godot-enemy-footing"

ROWS = [
    ("MLF-1", "the footing pass never runs", ZC,
     "\t_settle_enemy_footing()\n\t_evaluate_objectives()\n",
     "\t_evaluate_objectives()\n",
     "no enemy is built inside a solid"),
    ("MLF-2", "the fit test does not see solids", EF,
     "\t\tif who is StaticBody3D or who is RigidBody3D:\n"
     "\t\t\treturn \"inside %s\" % (who as Node).name\n",
     "\t\tif false:\n"
     "\t\t\treturn \"inside %s\" % (who as Node).name\n",
     "no enemy is built inside a solid"),
    ("MLF-3", "a reward pedestal moved after it enters the tree", ZC,
     "\t\t\treward.position = to_local(xform * (anchor + offset))\n"
     "\t\t\tadd_child(reward)\n",
     "\t\t\tadd_child(reward)\n"
     "\t\t\treward.global_position = xform * (anchor + offset)\n",
     "no enemy is built inside a solid"),
    ("MLF-4", "judged where it was placed, not where it lands", EF,
     "\t\tstand = Vector3(at.x, level + FOOT_CLEARANCE, at.z)\n",
     "\t\tstand = at\n",
     "every enemy the pass moved stood inside something"),
    ("MLF-9", "a head grazing a ceiling counted as inside", EF,
     "const HEAD_MARGIN := 0.05\n", "const HEAD_MARGIN := 0.0\n",
     "every enemy the pass moved stood inside something"),
    ("MLF-5", "a patrol beat drawn at random again", EN,
     "\t\tvar beat := _floored_beat()\n",
     "\t\tvar angle := randf() * TAU\n"
     "\t\tvar beat := post + Vector3(cos(angle), 0.0, sin(angle)) \\\n"
     "\t\t\t\t* Constants.ENEMY_PATROL_RADIUS\n",
     "every beat's path from the post keeps to the floor"),
    ("MLF-6", "the walk from where it stands not checked", EN,
     "\t\tif from_here.length() > EnemyFooting.WALK_SAMPLE \\\n",
     "\t\tif false and from_here.length() > EnemyFooting.WALK_SAMPLE \\\n",
     "every beat's path from the post keeps to the floor"),
    ("MLF-7", "the ledge guard off", EN,
     "\tif dir.length() > 0.05 and is_on_floor() \\\n"
     "\t\t\tand not bool(envelope.get(\"flying\", false)):\n",
     "\tif false:\n",
     "it stops on the slab"),
    ("MLF-8", "a walk home that meets a ledge keeps its old post", EN,
     "\tif returning:\n\t\treturning = false\n\t\tpost = global_position\n",
     "\tif returning:\n\t\tpass\n",
     "takes up a post on the slab"),
]


def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main() -> int:
    only = set(sys.argv[1:])
    lines, caught, total = [], 0, 0
    for rid, what, rel, old, new, expect in ROWS:
        if only and rid not in only:
            continue
        total += 1
        path = TREE / rel
        original = path.read_bytes()
        text = original.decode("utf-8")
        if text.count(old) != 1:
            lines.append(f"{rid} SETUP ERROR: anchor found {text.count(old)}"
                         f" times in {rel}")
            print(lines[-1], flush=True)
            continue
        before = hashlib.sha256(original).hexdigest()
        path.write_bytes(text.replace(old, new).encode("utf-8"))
        try:
            out = subprocess.run(["make", TARGET], cwd=TREE,
                                 capture_output=True, text=True, timeout=1500)
            log = out.stdout + out.stderr
        finally:
            path.write_bytes(original)
        restored = sha(path) == before
        fails = [l[6:] for l in log.splitlines() if l.startswith("FAIL: ")]
        errors = sum(1 for l in log.splitlines()
                     if l.lstrip().startswith("SCRIPT ERROR"))
        hit = any(expect in f for f in fails)
        caught += hit
        lines.append(f"{rid} [{rel.split('/')[-1]}] {what}")
        lines.append(f"    {'CAUGHT' if hit else 'NOT CAUGHT'} by \"{expect}\""
                     f" -- exit {out.returncode}, {len(fails)} failing check(s),"
                     f" {errors} script error(s); restored byte for byte:"
                     f" {restored}")
        for f in fails:
            lines.append(f"      FAIL: {f[:300]}")
        print("\n".join(lines[-(len(fails) + 2):]), flush=True)
    lines.append(f"\n{caught} of {total} sabotages caught")
    print(lines[-1])
    return 0 if caught == total else 1


if __name__ == "__main__":
    sys.exit(main())
