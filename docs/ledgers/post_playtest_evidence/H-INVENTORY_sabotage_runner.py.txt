def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def run_face(case: str) -> str:
    out = subprocess.run(GODOT + ["--equipment-face",
                                  f"--equipment-only={case}"],
                         capture_output=True, text=True, timeout=600,
                         cwd=TREE)
    return out.stdout + out.stderr


def main() -> int:
    lines = []
    caught = 0
    only = set(sys.argv[1:])
    rows = [r for r in ROWS if not only or r[0] in only]
    bridge_rows = [r for r in BRIDGE_ROWS if not only or r[0] in only]
    for rid, what, rel, old, new, case, expect in rows:
        path = TREE / rel
        before_bytes = path.read_bytes()
        before = sha(path)
        text = before_bytes.decode("utf-8")
        if text.count(old) != 1:
            lines.append(f"{rid} SETUP ERROR: patch anchor found "
                         f"{text.count(old)} times in {rel}")
            continue
        path.write_bytes(text.replace(old, new).encode("utf-8"))
        try:
            if rel == PROJECT:
                subprocess.run(["make", "godot-import"], cwd=TREE,
                               capture_output=True, timeout=600)
            log = run_face(case)
        finally:
            path.write_bytes(before_bytes)
            if rel == PROJECT:
                subprocess.run(["make", "godot-import"], cwd=TREE,
                               capture_output=True, timeout=600)
        restored = sha(path) == before
        fails = sorted({l[len("FAIL: "):] for l in log.splitlines()
                        if l.startswith("FAIL: ")})
        errors = sum(1 for l in log.splitlines() if "SCRIPT ERROR" in l)
        hit = any(expect in f for f in fails)
        caught += hit
        lines.append(f"{rid} [{rel.split('/')[-1]}] {what}")
        lines.append(f"    case {case}: {'CAUGHT' if hit else 'NOT CAUGHT'}"
                     f" -- {len(fails)} failing check(s), "
                     f"{errors} script error(s); restored byte for byte: "
                     f"{restored}")
        for f in fails:
            lines.append(f"      FAIL: {f}")
    # The fixture: the bridge's freshness test.
    for rid, what, rel in bridge_rows:
        path = TREE / rel
        before_bytes = path.read_bytes()
        before = sha(path)
        import json
        data = json.loads(before_bytes)
        for item in data["base"]["inventory"]["items"]:
            if item["component_id"] == "act_cinder":
                item["charges_left"] = 3
        path.write_text(json.dumps(data, indent=1, sort_keys=True) + "\n",
                        encoding="utf-8")
        try:
            out = subprocess.run(
                [sys.executable, "-m", "pytest", "-q",
                 "tests/test_equipment_fixture.py"],
                cwd=TREE / "bridge", capture_output=True, text=True,
                timeout=600)
        finally:
            path.write_bytes(before_bytes)
        restored = sha(path) == before
        hit = out.returncode != 0 and "matches_its_generator" in out.stdout
        caught += hit
        lines.append(f"{rid} [{rel.split('/')[-1]}] {what}")
        lines.append(f"    test_equipment_fixture.py: "
                     f"{'CAUGHT' if hit else 'NOT CAUGHT'}; restored byte "
                     f"for byte: {restored}")
        for l in out.stdout.splitlines():
            if l.startswith("FAILED"):
                lines.append(f"      {l}")
    total = len(rows) + len(bridge_rows)
    lines.append(f"\n{caught} of {total} sabotages caught")
    print("\n".join(lines))
    return 0 if caught == total else 1


if __name__ == "__main__":
    sys.exit(main())
