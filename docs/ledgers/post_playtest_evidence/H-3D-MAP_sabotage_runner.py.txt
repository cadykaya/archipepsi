def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main() -> int:
    only = set(sys.argv[1:])
    rows = [r for r in ROWS if not only or r[0] in only]
    lines, caught = [], 0
    for rid, what, rel, old, new, expect in rows:
        path = TREE / rel
        original = path.read_bytes()
        before = sha(path)
        text = original.decode("utf-8")
        if text.count(old) != 1:
            lines.append(f"{rid} SETUP ERROR: anchor found "
                         f"{text.count(old)} times in {rel}")
            continue
        path.write_bytes(text.replace(old, new).encode("utf-8"))
        try:
            out = subprocess.run(GODOT, capture_output=True, text=True,
                                 timeout=900, cwd=TREE)
            log = out.stdout + out.stderr
        finally:
            path.write_bytes(original)
        restored = sha(path) == before
        fails = sorted({l[6:] for l in log.splitlines()
                        if l.startswith("FAIL: ")})
        errors = sum(1 for l in log.splitlines() if "SCRIPT ERROR" in l)
        hit = any(expect in f for f in fails)
        caught += hit
        lines.append(f"{rid} [{rel.split('/')[-1]}] {what}")
        lines.append(f"    {'CAUGHT' if hit else 'NOT CAUGHT'} -- "
                     f"{len(fails)} failing check(s), {errors} script "
                     f"error(s); restored byte for byte: {restored}")
        for f in fails:
            lines.append(f"      FAIL: {f[:300]}")
    lines.append(f"\n{caught} of {len(rows)} sabotages caught")
    print("\n".join(lines))
    return 0 if caught == len(rows) else 1


if __name__ == "__main__":
    sys.exit(main())
